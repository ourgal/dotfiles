*** set corfu-auto

#+BEGIN_SRC emacs-lisp :tangle no
;; Define your list of trusted parent directories
(defvar my-trusted-directories
  '("~/.local/share/chezmoi")
  "List of directories where automatic completion is safe and allowed.")

(defun my-corfu-auto-if-trusted ()
  "Enable `corfu-auto` locally only if the current file is in a trusted directory."
  (when buffer-file-name
    (let ((file (expand-file-name buffer-file-name)))
      (when (seq-some (lambda (dir)
                        (string-prefix-p (expand-file-name dir) file))
                      my-trusted-directories)
        (setq-local corfu-auto t)))))

;; Add the check to `corfu-mode-hook`
(add-hook 'corfu-mode-hook #'my-corfu-auto-if-trusted)
#+END_SRC

The `evil-matchit' package is a extensible package that allows jumping

  ;; Set to nil if you prefer the advanced block-matching behavior in Python.

    (require 'evil-matchit-evil-setup))
